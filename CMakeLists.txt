cmake_minimum_required(VERSION 3.26)

# cudann: CUDA port of syclnn.  Two build flavours:
#   CUDA (default): nvcc + cuBLAS, CMAKE_CUDA_ARCHITECTURES from CUDANN_CUDA_ARCHS
#   HIP  (-DCUDANN_HIP=ON): the sources are converted with hipify-perl at configure
#        time and compiled with hipcc + hipBLAS (AMD GPUs, e.g. the RX 7900 XTX)
option(CUDANN_HIP "Build the HIPified variant with hipcc (AMD GPUs)" OFF)
set(CUDANN_CUDA_ARCHS "61;80" CACHE STRING "CUDA architectures (Pascal 61, Ampere 80)")
set(CUDANN_HIP_ARCHS "gfx1100" CACHE STRING "HIP offload architectures")
option(CUDANN_BUILD_BENCH "Build bench/train_bench" ON)
option(CUDANN_BUILD_PYTHON "Build the pybind11 module" ON)
option(CUDANN_FAST_MATH "nvcc --use_fast_math / hipcc -ffast-math (breaks parity)" OFF)
set(CUDANN_MODULE_NAME "_cudann" CACHE STRING "Python extension module name")

if(CUDANN_HIP)
  project(cudann VERSION 0.1.0 LANGUAGES CXX)
else()
  project(cudann VERSION 0.1.0 LANGUAGES CXX CUDA)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# ------------------------------------------------------------------ sources (possibly hipified)
set(_src_dir ${CMAKE_CURRENT_SOURCE_DIR})
set(_bindings ${_src_dir}/src/python_bindings.cu)
set(_bench ${_src_dir}/bench/train_bench.cu)
set(_include ${_src_dir}/include)

if(CUDANN_HIP)
  find_program(HIPIFY_PERL hipify-perl HINTS /opt/rocm/bin REQUIRED)
  set(_hip_dir ${CMAKE_BINARY_DIR}/hipified)
  file(MAKE_DIRECTORY ${_hip_dir}/include/cudann ${_hip_dir}/src ${_hip_dir}/bench)
  file(GLOB _headers ${_src_dir}/include/cudann/*.hpp)
  set(_hip_products)
  foreach(_f IN LISTS _headers _bindings _bench)
    file(RELATIVE_PATH _rel ${_src_dir} ${_f})
    string(REGEX REPLACE "\\.cu$" ".hip.cpp" _rel_out ${_rel})
    set(_out ${_hip_dir}/${_rel_out})
    # hipify-perl writes `#include <hipblas.h>`; ROCm installs it as <hipblas/hipblas.h>
    add_custom_command(OUTPUT ${_out}
      COMMAND ${HIPIFY_PERL} ${_f} -o ${_out}
      COMMAND ${CMAKE_COMMAND} -Dfile=${_out} -P ${_src_dir}/cmake/fix_hipblas_include.cmake
      DEPENDS ${_f} ${_src_dir}/cmake/fix_hipblas_include.cmake
      COMMENT "hipify ${_rel}" VERBATIM)
    list(APPEND _hip_products ${_out})
    # run once at configure time too, so that the generated headers exist for IDEs
    execute_process(COMMAND ${HIPIFY_PERL} ${_f} -o ${_out})
    execute_process(COMMAND ${CMAKE_COMMAND} -Dfile=${_out} -P ${_src_dir}/cmake/fix_hipblas_include.cmake)
  endforeach()
  add_custom_target(hipify_sources DEPENDS ${_hip_products})
  set(_bindings ${_hip_dir}/src/python_bindings.hip.cpp)
  set(_bench ${_hip_dir}/bench/train_bench.hip.cpp)
  set(_include ${_hip_dir}/include)
  configure_file(${_src_dir}/include/cudann/build_info.hpp.in ${_hip_dir}/include/cudann/build_info.hpp.in COPYONLY)
  find_package(hip REQUIRED)
  find_package(hipblas REQUIRED)
  set(CUDANN_PLATFORM "HIP")
  set(CUDANN_ARCH_STRING "${CUDANN_HIP_ARCHS}")
  set(CUDANN_BLAS_STRING "hipBLAS ${hipblas_VERSION}")
  string(REPLACE ";" "," _archs "${CUDANN_HIP_ARCHS}")
  add_library(cudann_backend INTERFACE)
  # hip::device adds -x hip and the HIP include directories (the sources keep a .cpp extension)
  target_link_libraries(cudann_backend INTERFACE hip::device hip::host roc::hipblas)
  target_compile_options(cudann_backend INTERFACE --offload-arch=${_archs} -Wno-unused-result)
  if(CUDANN_FAST_MATH)
    target_compile_options(cudann_backend INTERFACE -ffast-math)
  endif()
else()
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_ARCHITECTURES ${CUDANN_CUDA_ARCHS})
  set(CUDANN_PLATFORM "CUDA")
  set(CUDANN_ARCH_STRING "${CUDANN_CUDA_ARCHS}")
  set(CUDANN_BLAS_STRING "cuBLAS ${CUDAToolkit_VERSION}")
  add_library(cudann_backend INTERFACE)
  target_link_libraries(cudann_backend INTERFACE CUDA::cublas CUDA::cudart)
  target_compile_options(cudann_backend INTERFACE
    $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--extended-lambda;-lineinfo;-Wno-deprecated-gpu-targets>)
  if(CUDANN_FAST_MATH)
    target_compile_options(cudann_backend INTERFACE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
  endif()
endif()

# ------------------------------------------------------------------ the header-only library
add_library(cudann INTERFACE)
add_library(cudann::cudann ALIAS cudann)
target_include_directories(cudann INTERFACE $<BUILD_INTERFACE:${_include}> $<INSTALL_INTERFACE:include>)
target_link_libraries(cudann INTERFACE cudann_backend)

string(TOUPPER "${CMAKE_BUILD_TYPE}" _bt)
if(CUDANN_HIP)
  set(CUDANN_COMPILER_ID "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION} ${CMAKE_CXX_COMPILER} (hipcc)")
  set(CUDANN_COMPILE_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${_bt}} --offload-arch=${_archs}")
else()
  set(CUDANN_COMPILER_ID "nvcc ${CMAKE_CUDA_COMPILER_VERSION} ${CMAKE_CUDA_COMPILER} / host ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
  set(CUDANN_COMPILE_FLAGS "${CMAKE_CUDA_FLAGS} ${CMAKE_CUDA_FLAGS_${_bt}}")
endif()
string(STRIP "${CUDANN_COMPILE_FLAGS}" CUDANN_COMPILE_FLAGS)
find_package(Git QUIET)
set(CUDANN_GIT_SHA "unknown")
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C ${CMAKE_CURRENT_SOURCE_DIR} describe --always --dirty --tags
    OUTPUT_VARIABLE CUDANN_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
configure_file(${_src_dir}/include/cudann/build_info.hpp.in "${CMAKE_BINARY_DIR}/generated/cudann/build_info.hpp" @ONLY)
target_include_directories(cudann INTERFACE $<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/generated>)

# ------------------------------------------------------------------ Python module
if(CUDANN_BUILD_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG QUIET)
  if(NOT pybind11_FOUND)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
      OUTPUT_VARIABLE _pb11_dir OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(_pb11_dir)
      list(APPEND CMAKE_PREFIX_PATH "${_pb11_dir}")
      find_package(pybind11 CONFIG QUIET)
    endif()
  endif()
  if(NOT pybind11_FOUND)
    include(FetchContent)
    FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v3.1.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(pybind11)
  endif()

  if(CUDANN_HIP)
    # NO_EXTRAS: no LTO through the HIP toolchain (device code + thin LTO do not mix)
    pybind11_add_module(cudann_python MODULE NO_EXTRAS ${_bindings})
    add_dependencies(cudann_python hipify_sources)
  else()
    pybind11_add_module(cudann_python MODULE ${_bindings})
  endif()
  target_link_libraries(cudann_python PRIVATE cudann)
  target_compile_definitions(cudann_python PRIVATE CUDANN_MODULE_NAME=${CUDANN_MODULE_NAME})
  set_target_properties(cudann_python PROPERTIES
    OUTPUT_NAME "${CUDANN_MODULE_NAME}"
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    BUILD_RPATH "$ORIGIN"
    INSTALL_RPATH "$ORIGIN")
  if(NOT CUDANN_HIP)
    set_target_properties(cudann_python PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
  endif()
  if(SKBUILD_PROJECT_NAME)
    install(TARGETS cudann_python DESTINATION ${SKBUILD_PROJECT_NAME})
  endif()
endif()

# ------------------------------------------------------------------ C++ driver + smoke test
if(CUDANN_BUILD_BENCH)
  add_executable(train_bench ${_bench})
  if(CUDANN_HIP)
    add_dependencies(train_bench hipify_sources)
  endif()
  target_link_libraries(train_bench PRIVATE cudann)
  include(CTest)
  add_test(NAME smoke_train COMMAND train_bench --layers 8,16,4 --samples 64 --batch 16 --epochs 2 --check)
endif()

install(DIRECTORY include/ DESTINATION include)
